*** CRITICAL BEC REMINDERS ***
1. REMOVE ESC RED 5V WIRE AT S1: Pull the middle red wire out of the 50A ESC's
   3-pin servo plug and cover it with heatshrink. Only Signal and Ground go to S1.
   (Leaving the ESC's 5V/3A BEC connected will fight the FC's internal BEC.)
2. VTX BEC JUMPER = 9V: Set the FC's VTX BEC jumper to 9V so 3S voltage sag under
   full throttle never browns out the Rush Tank Solo 1.6W VTX.
3. SERVO BEC JUMPER (Vx) = 5V or 6V: Match your 3 servos' rated voltage.


5. INAV CLI CONFIGURATION (CANARD + ELEVONS + AUTOLAUNCH)
--------------------------------------------------------------------------------
# --- 1. SERIAL PORTS (UART1=SmartAudio, UART2=ELRS RX, UART6=GPS) ---
serial 0 2048 115200 115200 0 115200
serial 1 64 115200 115200 0 115200
serial 5 2 115200 115200 0 115200

# --- 2. RECEIVER, GPS & COMPASS ---
set receiver_type = SERIAL
set serialrx_provider = CRSF
feature GPS
set gps_provider = UBLOX
set gps_sbas_mode = AUTO
set mag_hardware = QMC5883
set align_mag = CW270FLIP

# --- 3. MOTOR & SERVO MIXER (S1=Motor, S2=Left Elevon, S3=Right Elevon, S4=Canard) ---
mmix reset
mmix 0 1.000 0.000 0.000 0.000

smix reset
# S2 (Servo 1) = Left Wing Elevon (Roll + Pitch)
smix 0 1 0 100 0
smix 1 1 1 100 0
# S3 (Servo 2) = Right Wing Elevon (Roll + Pitch)
smix 2 2 0 100 0
smix 3 2 1 -100 0
# S4 (Servo 3) = Front Canard (Pitch Only — moves opposite rear elevons)
smix 4 3 1 -100 0

# --- 4. VTX SMARTAUDIO, BUNGEE AUTOLAUNCH & FAILSAFE RTH ---
set vtx_band = 5
set vtx_channel = 1
set vtx_power = 4
set nav_fw_launch_velocity = 300
set nav_fw_launch_accel = 1863
set nav_fw_launch_max_angle = 45
set nav_fw_launch_motor_delay = 500
set nav_fw_launch_spinup_time = 100
set nav_fw_launch_min_time = 0
set nav_fw_launch_timeout = 5000
set nav_fw_launch_max_altitude = 5000
set nav_fw_launch_climb_angle = 18
set nav_fw_launch_thr = 1700
set nav_fw_launch_idle_thr = 1000
set failsafe_procedure = RTH

save
